# Database ER Diagram

```mermaid
erDiagram
    %% Users and Authentication
    USER ||..|| USER_PROFILE : has
    USER ||..|| SESSION : creates
    USER ||..|| API_KEY : owns
    USER ||..|| WATCHLIST : owns
    USER ||..|| RECOMMENDATION : receives
    USER ||..|| RECOMMENDATION_FEEDBACK : provides
    USER ||..|| ALERT : creates
    USER ||..|| ANALYTICS_EVENT : generates
    USER ||..|| AUDIT_LOG : triggers
    
    %% User Profile
    USER_PROFILE ||..|| USER : belongs to
    
    ;; Sessions
    SESSION ||..|| USER : belongs to
    
    ;; API Keys
    API_KEY ||..|| USER : owned by
    
    ;; Watchlists
    WATCHLIST ||..|| USER : owned by
    WATCHLIST }|..|| WATCHLIST_ITEM : contains
    WATCHLIST_ITEM ||..|| WATCHLIST : belongs to
    WATCHLIST_ITEM }|..|| MARKET_DATA_CACHE : references
    
    ;; Recommendations
    RECOMMENDATION ||..|| USER : belongs to
    RECOMMENDATION ||..|| RECOMMENDATION_FEEDBACK : has
    RECOMMENDATION }|..|| MARKET_DATA_CACHE : based on
    RECOMMENDATION_FEEDBACK ||..|| USER : given by
    RECOMMENDATION_FEEDBACK ||..|| RECOMMENDATION : for
    
    ;; Alerts
    ALERT ||..|| USER : created by
    ALERT }|..|| ALERT_TRIGGER : triggers
    ALERT_TRIGGER ||..|| ALERT : belongs to
    ALERT_TRIGGER }|..|| MARKET_DATA_CACHE : evaluated against
    
    ;; Market Data
    MARKET_DATA_CACHE }|..|| WATCHLIST_ITEM : referenced by
    MARKET_DATA_CACHE }|..|| RECOMMENDATION : used for
    MARKET_DATA_CACHE }|..|| ALERT_TRIGGER : used for
    MARKET_DATA_CACHE }|..|| ANALYTICS_EVENT : referenced in
    
    ;; News and Sentiment
    NEWS_ARTICLE }|..|| ANALYTICS_EVENT : influences
    NEWS_ARTICLE }|..|| USER : read by (optional tracking)
    
    ;; Analytics
    ANALYTICS_EVENT ||..|| USER : generated by
    ANALYTICS_EVENT }|..|| MARKET_DATA_CACHE : related to
    ANALYTICS_EVENT }|..|| NEWS_ARTICLE : related to
    
    ;; Audit Log
    AUDIT_LOG ||..|| USER : triggered by
    AUDIT_LOG }|..|| USER : affected user (optional)
    AUDIT_LOG }|..|| API_KEY : used (optional)
    AUDIT_LOG }|..|| WATCHLIST : modified (optional)
    AUDIT_LOG }|..|| ALERT : modified (optional)
    AUDIT_LOG }|..|| RECOMMENDATION : accessed (optional)
    
    %% Entity Definitions
    USER {
        uuid id PK
        string email
        string password_hash
        string first_name
        string last_name
        enum status
        datetime created_at
        datetime updated_at
        datetime last_login_at
    }
    
    USER_PROFILE {
        uuid id PK
        uuid user_id FK
        string avatar_url
        string bio
        json preferences
        json settings
        datetime created_at
        datetime updated_at
    }
    
    SESSION {
        uuid id PK
        uuid user_id FK
        string token_hash
        datetime expires_at
        string user_agent
        string ip_address
        datetime created_at
    }
    
    API_KEY {
        uuid id PK
        uuid user_id FK
        string provider
        string encrypted_key
        string encrypted_secret
        datetime created_at
        datetime last_used_at
        datetime expires_at
    }
    
    WATCHLIST {
        uuid id PK
        uuid user_id FK
        string name
        string description
        datetime created_at
        datetime updated_at
    }
    
    WATCHLIST_ITEM {
        uuid id PK
        uuid watchlist_id FK
        string symbol
        string exchange
        datetime added_at
    }
    
    RECOMMENDATION {
        uuid id PK
        uuid user_id FK
        string symbol
        string timeframe
        enum recommendation_type
        float confidence
        json reasoning
        datetime created_at
        datetime expires_at
    }
    
    RECOMMENDATION_FEEDBACK {
        uuid id PK
        uuid recommendation_id FK
        uuid user_id FK
        enum feedback_type
        enum trade_result
        float profit_loss_pct
        datetime created_at
    }
    
    ALERT {
        uuid id PK
        uuid user_id FK
        string name
        string description
        json conditions
        boolean is_active
        datetime created_at
        datetime updated_at
        datetime last_triggered_at
    }
    
    ALERT_TRIGGER {
        uuid id PK
        uuid alert_id FK
        datetime triggered_at
        float price_at_trigger
        json metadata
    }
    
    MARKET_DATA_CACHE {
        uuid id PK
        string symbol
        string timeframe
        enum data_type
        json data
        datetime expires_at
    }
    
    NEWS_ARTICLE {
        uuid id PK
        string source
        string title
        string text
        string url
        datetime published_at
        float sentiment_score
        json entities
        json symbols
        datetime created_at
    }
    
    ANALYTICS_EVENT {
        uuid id PK
        uuid user_id FK
        uuid session_id FK
        string event_type
        json properties
        datetime timestamp
    }
    
    AUDIT_LOG {
        uuid id PK
        uuid user_id FK
        string action
        string resource_type
        uuid resource_id
        json changes
        string ip_address
        string user_agent
        datetime created_at
    }
```
